    _devon_init(self, root, **kwargs)
    self._devon_store = kwargs['store']
    self._devon_paths = kwargs['settings'].runtime.paths
    self._devon_clock = kwargs['clock']

    from ovcaq.runtime.session import record_state_failure
    def notice(site, detail):
        def update(state):
            state['control']['attention'] = {'site': site, 'detail': detail, 'created_at': self._devon_clock.now().isoformat().replace('+00:00', 'Z')}
