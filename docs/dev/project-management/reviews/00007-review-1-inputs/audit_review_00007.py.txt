"""Read-only workflow checks and durable evidence capture for review handoff."""

import hashlib
import json
import re
import shutil
import subprocess
import sys
from pathlib import Path

ROOT = Path('/Users/bob/git/src/github.com/buvis/ovcaq')
PLUGIN = Path('/Users/bob/.claude/plugins/cache/buvis-plugins/autopilot/0.6.0')
REV = ROOT / 'docs/dev/project-management/reviews'
AUTO = ROOT / 'docs/dev/project-management/autopilot'
TMP = ROOT / 'docs/dev/tmp'
STEM = '00007-supervisor-loop-v1'
sys.path.insert(0, str(PLUGIN / 'skills/run-autopilot'))
from cli.schema import validate

state = json.loads((AUTO / 'state.json').read_text())
validate(state)
assert state['phase'] == state['next_phase'] == 'review' and state['cycle'] == 1
assert not state.get('pause_reason') and not state.get('stall_reason')
assert 'review' not in state['phases_completed']
tasks = state['tasks']
assert state['tasks_total'] == len(tasks) == 15
assert state['tasks_completed'] == sum(t['status'] == 'completed' for t in tasks) == 8
assert state['rework_task_ids'] == [str(i) for i in range(9, 16)]
assert all(t['status'] == 'pending' for t in tasks if t['id'] in state['rework_task_ids'])
task_map = json.loads((REV / '00007-review-1-task-map.json').read_text())
decisions = json.loads((REV / '00007-review-1-decisions.json').read_text())
records = state['autonomous_decisions'] + state['deferred_decisions']
for d in decisions:
    assert any(r.get('finding_id') == d['id'] and r.get('cycle') == 1 for r in records), d['id']
    if d['action'] == 'fix':
        task = next(t for t in tasks if t['id'] == task_map[d['id']])
        exact = f"{d['severity']} {d['issue']} | File: {d['file']} | Consensus: {d['consensus']}"
        assert exact in task['description'], d['id']
review = (REV / f'{STEM}-review-1.md').read_text()
expected = {'alice': ('R', [1, 2, 3, 4, 6, 7, 8, 9, 10, 11, 12, 13]),
            'blake': ('B', list(range(1, 20))), 'bob': ('D', list(range(1, 6)))}
rubrics = {}
for name, (prefix, ids) in expected.items():
    raw = (REV / f'00007-review-1-{name}.txt').read_text()
    assert raw in review
    lines = re.findall(rf'^{prefix}(\d+): (pass|fail)$', raw, re.M)
    assert sorted(int(n) for n, _ in lines) == ids
    rubrics[name] = dict(lines)
assert 'codex_thread_id:' not in review
receipt = json.loads((REV / '00007-review-1-bob.txt.receipt.json').read_text())
for key, digest in [('output', 'sha256'), ('raw', 'raw_sha256')]:
    assert hashlib.sha256(Path(receipt[key]).read_bytes()).hexdigest() == receipt[digest]
assert state['doubts_rubric_verdicts'] == [
    {'rule_id': f'D{i}', 'verdict': 'pass', 'source': 'claude'} for i in range(1, 6)
]
queue = json.loads((REV / f'{STEM}-checks-1.json').read_text())
assert len(queue) == 1 and 'result' not in queue[0]
assert queue[0]['command'] == 'UV_CACHE_DIR=/private/tmp/ovcaq-final-cache uv run pytest'
ledger = json.loads((REV / f'{STEM}-ledger.json').read_text())
assert len(ledger) == 10
sink = json.loads((AUTO / 'deferred/202609062243-deferred.json').read_text())
owned = [item for item in sink['items'] if item.get('finding_id') in {'F02', 'F06', 'F23'} and item.get('prd') == STEM + '.md']
assert {r['finding_id'] for r in owned} == {'F02', 'F06', 'F23'}, owned
(REV / '00007-review-1-deferrals.json').write_text(json.dumps(owned, indent=2) + '\n')

hook_results = {}
for name, input_text in [
    ('validate_state_json_hook.py', json.dumps({'tool_input': {'file_path': str(AUTO / 'state.json')}})),
    ('review_coverage_hook.py', '{}'),
]:
    result = subprocess.run(['python3', str(PLUGIN / 'skills/run-autopilot/scripts' / name)],
                            input=input_text, text=True, capture_output=True, cwd=ROOT)
    assert result.returncode == 0, result.stderr
    hook_results[name] = {'exit': result.returncode, 'stderr': result.stderr}

# Preserve the submitted package and the probe as inert evidence, then remove
# only this session's reproducible scratch. Existing design/build scratch stays.
inputs = REV / '00007-review-1-inputs'
for path in TMP.glob('*-00007-1-20261002.*'):
    target = inputs / path.name
    shutil.copyfile(path, target)
    assert target.read_bytes() == path.read_bytes()
for name in ('prepare_review_00007.py', 'consolidate_review_00007.py',
             'decide_review_00007.py', 'publish_review_00007.py',
             'probe_review_00007.py', 'audit_review_00007.py'):
    shutil.copyfile(TMP / name, inputs / (name + '.txt'))
audit = {
    'schema': 'pass', 'counts': {'completed': 8, 'total': 15},
    'phase': state['phase'], 'next_phase': state['next_phase'], 'cycle': 1,
    'rework_task_ids': state['rework_task_ids'],
    'coverage': {'findings': 32, 'fixed_in_queued_tasks': len(task_map),
                 'deferred': 3, 'discarded': 7, 'routed_to_verification': 1},
    'raw_rubrics': rubrics, 'bob_receipt_hashes': 'pass',
    'hook_results': hook_results,
    'note': 'Task dependency payload initially used string IDs; schema rejected it without mutating state. Corrected blocked_by to integers. No production source changed.',
}
(REV / '00007-review-1-audit.json').write_text(json.dumps(audit, indent=2) + '\n')
print(json.dumps(audit, indent=2))
